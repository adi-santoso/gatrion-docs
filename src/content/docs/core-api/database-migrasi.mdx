<CodeBlock code={`generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id        String   @id @default(cuid())
  email     String   @unique
  password  String
  name      String
  role      Role     @default(USER)
  orders    Order[]
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([email])
}

model Order {
  id          String   @id @default(cuid())
  userId      String
  user        User     @relation(fields: [userId], references: [id])
  total       Decimal  @db.Decimal(12, 2)
  status      OrderStatus @default(PENDING)
  items       OrderItem[]
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@index([userId])
  @@index([status])
}

enum Role {
  ADMIN
  USER
}

enum OrderStatus {
  PENDING
  PAID
  SHIPPED
  DELIVERED
  CANCELLED
}`} lang="bash" title="prisma/schema.prisma" />

<p class="mt-3 leading-relaxed">Perintah untuk membuat dan menjalankan migrasi:</p>

<CodeBlock code={`# generate migration dari schema changes
npx prisma migrate dev --name add_orders_table

# apply migrations ke database
npx prisma migrate deploy

# reset database (DESTRUCTIVE)
npx prisma migrate reset

# status migrasi
npx prisma migrate status`} lang="bash" title="terminal" />

<Callout type="warning">
  **Hati-hati:** <code class="rounded-md bg-amber-500/10 px-1.5 py-0.5 font-mono text-[12px] text-amber-600 dark:text-amber-300">migrate reset</code> akan menghapus semua data. Gunakan hanya di development.
</Callout>

<p class="mt-3 leading-relaxed">Seed database dengan data dummy untuk development:</p>

<CodeBlock code={`// prisma/seed.ts
import { PrismaClient } from '@prisma/client';
import * as bcrypt from 'bcrypt';

const prisma = new PrismaClient();

async function main() {
  // Create admin user
  const admin = await prisma.user.create({
    data: {
      email: 'admin@gatrion.com',
      password: await bcrypt.hash('admin123', 10),
      name: 'Admin Gatrion',
      role: 'ADMIN',
    },
  });

  // Create sample orders
  await prisma.order.create({
    data: {
      userId: admin.id,
      total: 150000,
      status: 'PENDING',
    },
  });

  console.log('Seed completed!');
}

main();`} lang="bash" title="prisma/seed.ts" />

<CodeBlock code={`# run seed
npm run db:seed`} lang="bash" title="terminal" />

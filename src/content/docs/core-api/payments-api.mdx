Content-Type: application/json

  "orderId": "order_456",

  "channel": "BCVA"  // Bank Transfer VA
}`} lang="bash" title="Request" />

  <CodeBlock code={`{
  "statusCode": 201,
  "data": {

    "orderId": "order_456",
    "amount": "225000.00",

    "status": "PENDING",
    "redirectUrl": "https://app.sandbox.midtrans.com/...",

  }
}`} lang="json" title="Response 201" />

<section class="animate-fade-up [animation-delay:.15s]">
  <h2 id="get-payment" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
    <a href="#get-payment" class="flex items-center gap-2">Get Payment by ID <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
  </h2>

  <div class="mt-3 flex items-center gap-2">
    <span class="rounded-md bg-emerald-500/10 px-2 py-0.5 font-mono text-[11px] font-bold text-emerald-500">GET</span>
    <code class="font-mono text-[13px] text-zinc-600 dark:text-zinc-300">/api/payments/:id</code>
  </div>

  <CodeBlock code={`{
  "statusCode": 200,
  "data": {
    "id": "pay_789",
    "orderId": "order_456",
    "amount": "225000.00",
    "method": "MIDTRANS",
    "status": "SETTLED",
    "paidAt": "2025-01-30T12:30:00.000Z",
    "createdAt": "2025-01-30T12:00:00.000Z"
  }
}`} lang="json" title="Response 200" />
</section>

<section class="animate-fade-up [animation-delay:.2s]">
  <h2 id="refund" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
    <a href="#refund" class="flex items-center gap-2">Refund Payment <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
  </h2>

  <div class="mt-3 flex items-center gap-2">
    <span class="rounded-md bg-amber-500/10 px-2 py-0.5 font-mono text-[11px] font-bold text-amber-500">PUT</span>
    <code class="font-mono text-[13px] text-zinc-600 dark:text-zinc-300">/api/payments/:id/refund</code>
  </div>

  <CodeBlock code={`PUT /api/payments/pay_789/refund
Authorization: Bearer <token>
Content-Type: application/json

{
  "reason": "Customer cancelled order",
  "amount": 225000  // partial refund supported
}`} lang="bash" title="Request" />

  <CodeBlock code={`{
  "statusCode": 200,
  "data": {
    "id": "pay_789",
    "status": "REFUNDED",
    "refundAmount": "225000.00",
    "refundedAt": "2025-01-30T14:00:00.000Z"
  }
}`} lang="json" title="Response 200" />

  <Callout type="warning">
    **Refund hanya bisa dilakukan pada payment dengan status SETTLED.** Refund partial didukung dengan mengirim <code class="rounded-md bg-amber-500/10 px-1.5 py-0.5 font-mono text-[12px] text-amber-600 dark:text-amber-300">amount</code> kurang dari total.
  </Callout>
</section>

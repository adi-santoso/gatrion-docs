<section class="animate-fade-up [animation-delay:.2s]">
<h2 id="environment" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
  <a href="#environment" class="flex items-center gap-2">Environment Variables <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<CodeBlock code={`# copy env file
cp .env.example .env.local`} lang="bash" title="terminal" />

<CodeBlock code={`# .env.local
VITE_API_URL=http://localhost:8000/api
VITE_API_TIMEOUT=30000

# Auth
VITE_AUTH_TOKEN_KEY=gatrion_token
VITE_AUTH_REFRESH_KEY=gatrion_refresh

# Features
VITE_ENABLE_ANALYTICS=true
VITE_ENABLE_NOTIFICATIONS=true

# External
VITE_SENTRY_DSN=
VITE_MIDTRANS_CLIENT_KEY=SB-Mid-client-xxx`} lang="bash" title=".env.local" />

<Callout type="warning">
  **Penting:** Pastikan Core API sudah berjalan di <code class="rounded-md bg-amber-500/10 px-1.5 py-0.5 font-mono text-[12px] text-amber-600 dark:text-amber-300">http://localhost:8000</code> sebelum start dashboard.
</Callout>
</section>

  icon: string;

      </CardHeader>

<CodeBlock code={`// src/hooks/use-dashboard-stats.ts
export function useDashboardStats() {
  return useQuery({
    queryKey: ['dashboard-stats'],
    queryFn: () => apiClient.get('/analytics/dashboard').then(res => res.data.data),
    refetchInterval: 30000, // auto-refresh 30s
    refetchIntervalInBackground: false,
  });
}

export function useRevenueChart(period: 'daily' | 'weekly' | 'monthly') {
  return useQuery({
    queryKey: ['revenue-chart', period],
    queryFn: () => apiClient.get('/analytics/revenue', { params: { period } })
      .then(res => res.data.data),
  });
}`} lang="bash" title="use-dashboard-stats.ts" />
</section>

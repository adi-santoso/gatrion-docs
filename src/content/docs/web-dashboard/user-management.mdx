<CodeBlock code={`// src/hooks/use-users.ts
export function useCreateUser() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (data: CreateUserDto) => apiClient.post('/users', data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['users'] });
      toast.success('User berhasil dibuat');
    },
    onError: (err) => toast.error(err.response?.data?.message || 'Gagal membuat user'),
  });
}

export function useUpdateUser() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: ({ id, data }: { id: string; data: UpdateUserDto }) => 
      apiClient.put(\`/users/\${id}\`, data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['users'] });
      toast.success('User berhasil diupdate');
    },
  });
}`} lang="bash" title="use-users.ts" />

<CodeBlock code={`// src/hooks/use-permissions.ts
export function usePermissions() {
  const user = useAuthStore(s => s.user);
  
  return {
    canViewUsers: user?.role === 'ADMIN',
    canEditUser: (targetId: string) => 
      user?.role === 'ADMIN' || user?.id === targetId,
    canDeleteUser: user?.role === 'ADMIN',
    canViewAnalytics: user?.role === 'ADMIN',
  };
}`} lang="bash" title="use-permissions.ts" />

<CodeBlock code={`export function useBulkDeleteUsers() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (ids: string[]) => {
      await Promise.all(ids.map(id => apiClient.delete(\`/users/\${id}\`)));
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['users'] });
      toast.success(\`\${ids.length} user dihapus\`);
    },
  });
}`} lang="bash" title="use-bulk-actions.ts" />

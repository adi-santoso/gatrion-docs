<CodeBlock code={`// src/services/api-client.ts
import axios from 'axios';

export const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_URL,
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' },
});

// Request interceptor - attach token
apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('gatrion_token');
  if (token) {
    config.headers.Authorization = \`Bearer \${token}\`;
  }
  return config;
});

// Response interceptor - handle errors & refresh
apiClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    if (error.response?.status === 401) {
      const refreshed = await refreshToken();
      if (refreshed) {
        return apiClient(error.config); // retry
      }
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);`} lang="bash" title="api-client.ts" />
</section>

<p class="mt-3 leading-relaxed">Menggunakan <strong>Zustand</strong> untuk global state dan <strong>TanStack Query</strong> untuk server state:</p>

<CodeBlock code={`// src/stores/auth-store.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface AuthState {
  user: User | null;
  token: string | null;
  setAuth: (user: User, token: string) => void;
  logout: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      user: null,
      token: null,
      setAuth: (user, token) => set({ user, token }),
      logout: () => set({ user: null, token: null }),
    }),
    { name: 'gatrion-auth' }
  )
);`} lang="bash" title="auth-store.ts" />

<CodeBlock code={`// src/hooks/use-users.ts
import { useQuery } from '@tanstack/react-query';
import { apiClient } from '@/services/api-client';

export function useUsers(page = 1) {
  return useQuery({
    queryKey: ['users', page],
    queryFn: () => apiClient.get('/users', { params: { page } })
      .then(res => res.data),
  });
}`} lang="bash" title="use-users.ts" />
</section>

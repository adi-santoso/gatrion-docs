<CodeBlock code={`// main.dart
void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  
  await Firebase.initializeApp();
  
  // Pass Crashlytics to SDK
  await GatrionSDK.initialize(
    apiKey: 'YOUR_API_KEY',
    environment: Environment.production,
    crashlytics: FirebaseCrashlytics.instance,
    enableCrashReporting: true,
  );
  
  // Catch all errors
  FlutterError.onError = FirebaseCrashlytics.instance.recordFlutterFatalError;
  
  // Catch async errors
  PlatformDispatcher.instance.onError = (error, stack) {
    FirebaseCrashlytics.instance.recordError(error, stack);
    return true;
  };
  
  runApp(MyApp());
}`} lang="bash" title="main.dart" />
</section>

<CodeBlock code={`// SDK auto-tracks network errors via interceptor
// Configure which errors to report:
await GatrionSDK.initialize(
  apiKey: apiKey,
  crashlytics: FirebaseCrashlytics.instance,
  errorTrackingConfig: ErrorTrackingConfig(
    reportNetworkErrors: true,      // report 5xx errors
    reportTimeouts: true,           // report timeout errors
    reportConnectionErrors: true,   // report no internet
    excludeStatusCodes: [401, 403], // don't report auth errors
    sampleRate: 1.0,                // 100% in dev, 0.1 in prod
  ),
);

// Custom network error handler
GatrionSDK.client.addResponseInterceptor((response) {
  if (response.statusCode >= 500) {
    GatrionSDK.errors.record(
      'Server error: \${response.statusCode}',
      reason: 'API request failed',
      properties: {
        'endpoint': response.request.path,
        'method': response.request.method,
        'status': response.statusCode,
      },
    );
  }
  return response;
});`} lang="bash" title="network-errors.dart" />

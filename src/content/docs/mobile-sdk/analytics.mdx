  'price': 75000,
  'quantity': 2,
});

  'total_items': 3,
  'revenue': 225000,
  'currency': 'IDR',
  'payment_method': 'bank_transfer',
});`} lang="bash" title="track-events.dart" />

});

// Auto-track with route observer
MaterialApp(
  navigatorObservers: [
    GatrionSDK.analytics.routeObserver(),
  ],
  routes: {
    '/': (ctx) => HomePage(),
    '/products': (ctx) => ProductsPage(),
    '/cart': (ctx) => CartPage(),
  },

<section class="animate-fade-up [animation-delay:.2s]">
<h2 id="user-properties" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
  <a href="#user-properties" class="flex items-center gap-2">User Properties <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<CodeBlock code={`// Set user properties (auto sent with every event)
await GatrionSDK.analytics.setUserProperty('plan', 'premium');
await GatrionSDK.analytics.setUserProperty('signup_date', '2025-01-30');
await GatrionSDK.analytics.setUserProperty('total_orders', '15');

// Increment numeric property
await GatrionSDK.analytics.incrementUserProperty('total_orders');

// Get user properties
final props = await GatrionSDK.analytics.getUserProperties();
print('Plan: \${props['plan']}');`} lang="bash" title="user-properties.dart" />
</section>

<section class="animate-fade-up [animation-delay:.25s]">
<h2 id="funnel" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
  <a href="#funnel" class="flex items-center gap-2">Funnel Tracking <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<CodeBlock code={`// Start a funnel
final funnel = GatrionSDK.analytics.startFunnel('checkout_flow');

funnel.step('view_cart');
// user views cart

funnel.step('add_shipping_info', properties: {
  'shipping_method': 'same_day',
  'cost': 25000,
});
// user adds shipping

funnel.step('select_payment', properties: {
  'method': 'bank_transfer',
});
// user selects payment

funnel.complete(properties: {
  'order_id': 'order_456',
  'total': 150000,
});
// funnel completed

// Or abandon
// funnel.abandon(reason: 'user_closed_app');`} lang="bash" title="funnel.dart" />
</section>

<CodeBlock code={`// Sessions auto-start when SDK initializes
// Default: 30 min timeout

// Get session info
final session = GatrionSDK.analytics.getSession();
print('Session ID: \${session.id}');
print('Started: \${session.startTime}');
print('Events: \${session.eventCount}');

// Manually start new session
await GatrionSDK.analytics.startNewSession();

// Set session timeout (default: 30 min)
await GatrionSDK.initialize(
  apiKey: apiKey,
  sessionTimeout: Duration(minutes: 15),
);`} lang="bash" title="session.dart" />
</section>

final products = await GatrionSDK.client.get<List<Product>>(
  '/products',
  parser: (json) => (json as List)
      .map((e) => Product.fromJson(e))
      .toList(),
);

// Single object
final user = await GatrionSDK.client.get<User>(
  '/users/me',
  parser: (json) => User.fromJson(json as Map<String, dynamic>),
);

// Paginated response
final page = await GatrionSDK.client.get<PaginatedResponse<Product>>(
  '/products?page=1',
  parser: (json) => PaginatedResponse.fromJson(
    json, 
    itemsParser: (items) => items.map((e) => Product.fromJson(e)).toList(),
  ),
);`} lang="bash" title="typed-requests.dart" />

<CodeBlock code={`try {
  final result = await GatrionSDK.client.post('/orders', body: data);
  print('Order created: \${result['id']}');
} on GatrionException catch (e) {
  switch (e.type) {
    case GatrionErrorType.network:
      showSnackBar('No internet connection');
      break;
    case GatrionErrorType.unauthorized:
      // token expired, SDK auto-refreshes
      break;
    case GatrionErrorType.validation:
      showFormErrors(e.fieldErrors);
      break;
    case GatrionErrorType.server:
      showSnackBar('Server error: \${e.message}');
      break;
  }
} catch (e) {
  showSnackBar('Unexpected error: $e');
}`} lang="bash" title="error-handling.dart" />
</section>

<CodeBlock code={`// Add request interceptor
GatrionSDK.client.addRequestInterceptor((request) {
  // add custom header
  request.headers['X-App-Version'] = '1.0.0';
  request.headers['X-Platform'] = Platform.operatingSystem;
  return request;
});

// Add response interceptor
GatrionSDK.client.addResponseInterceptor((response) {
  // log response time
  final duration = response.duration;
  debugPrint('\${response.request.path} took \${duration}ms');
  return response;
});`} lang="bash" title="interceptors.dart" />
</section>

final transactions = await GatrionSDK.transactions.list(

  endDate: DateTime.now(),
);

for (final tx in transactions.data) {
  print('\${tx.id} | \${tx.amount} | \${tx.status} | \${tx.createdAt}');
}

// Pagination info
print('Total: \${transactions.meta.total}');
print('Pages: \${transactions.meta.totalPages}');`} lang="bash" title="history.dart" />

<CodeBlock code={`// Get single transaction
final tx = await GatrionSDK.transactions.get('pay_789');

print('ID: \${tx.id}');
print('Amount: \${tx.amount}');
print('Status: \${tx.status}');
print('Method: \${tx.method}');
print('Order ID: \${tx.orderId}');
print('Created: \${tx.createdAt}');
if (tx.paidAt != null) {
  print('Paid: \${tx.paidAt}');
}

// Get receipt URL
final receiptUrl = tx.receiptUrl;
if (receiptUrl != null) {
  await GatrionSDK.utils.openUrl(receiptUrl);
}`} lang="bash" title="detail.dart" />
</section>

<CodeBlock code={`// Filter by multiple statuses
final transactions = await GatrionSDK.transactions.list(
  statuses: [
    TransactionStatus.settled,
    TransactionStatus.pending,
  ],
  method: PaymentMethod.bankTransfer,
  minAmount: 50000,
  maxAmount: 1000000,
  sort: TransactionSort.createdAtDesc,
);

// Search by ID or order ID
final results = await GatrionSDK.transactions.search('order_456');`} lang="bash" title="filtering.dart" />
</section>

final file = await GatrionSDK.utils.saveFile(

  csvData,
);
await GatrionSDK.utils.shareFile(file);`} lang="bash" title="export.dart" />

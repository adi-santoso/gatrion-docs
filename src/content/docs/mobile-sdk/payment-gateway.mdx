<CodeBlock code={`// Get payment status
final status = await GatrionSDK.payments.getStatus('pay_789');
print('Status: \${status.status}');
// pending | settled | failed | refunded

// Listen to real-time status updates
GatrionSDK.payments.statusStream('pay_789').listen((status) {
  switch (status) {
    case PaymentStatus.pending:
      showLoading('Waiting for payment...');
      break;
    case PaymentStatus.settled:
      hideLoading();
      showSuccess('Payment successful!');
      Navigator.pushReplacementNamed(context, '/order/success');
      break;
    case PaymentStatus.failed:
      hideLoading();
      showError('Payment failed');
      break;
  }
});`} lang="bash" title="status.dart" />
</section>

  refreshOrders();
});

  showPaymentFailedDialog();
});`} lang="bash" title="webhook.dart" />

<h2 id="initialization" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
  <a href="#initialization" class="flex items-center gap-2">Initialization <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<p class="mt-3 leading-relaxed">Initialize SDK di <code class="rounded-md bg-zinc-100 px-1.5 py-0.5 font-mono text-[13px] dark:bg-white/[.06]">main.dart</code>:</p>

<CodeBlock code={`import 'package:gatrion_sdk/gatrion_sdk.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  
  await GatrionSDK.initialize(
    apiKey: 'YOUR_API_KEY',
    environment: Environment.production,
  );
  
  runApp(MyApp());
}`} lang="bash" title="main.dart" />

<Callout type="info">
  **Security:** Jangan hardcode API key di production. Gunakan environment variables atau secure storage.
</Callout>
</section>

<section class="animate-fade-up [animation-delay:.2s]">
<h2 id="basic-usage" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
  <a href="#basic-usage" class="flex items-center gap-2">Basic Usage <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<p class="mt-3 leading-relaxed">Contoh authentication dan fetch user data:</p>

<CodeBlock code={`// Login user
final result = await GatrionSDK.auth.login(
  email: 'user@example.com',
  password: 'password123',
);

if (result.success) {
  print('Token: \${result.token}');
}

// Get current user
final user = await GatrionSDK.users.getCurrentUser();
print('Hello, \${user.name}!');`} lang="bash" title="example.dart" />
</section>

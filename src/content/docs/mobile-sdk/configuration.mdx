}

Future<void> main() async {

  await GatrionSDK.initialize(
    apiKey: dotenv.env['GATRION_API_KEY']!,
    environment: dotenv.env['GATRION_ENV'] == 'production'
        ? Environment.production
        : Environment.sandbox,
  );
  
  runApp(MyApp());
}`} lang="typescript" title="main.dart" />

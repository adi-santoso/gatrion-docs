<p class="mt-3 leading-relaxed">Plugin adalah class yang implements <code class="rounded-md bg-zinc-100 px-1.5 py-0.5 font-mono text-[13px] dark:bg-white/[.06]">GatrionPlugin</code> dan dapat didaftarkan ke SDK:</p>

<CodeBlock code={`import 'package:gatrion_sdk/gatrion_sdk.dart';

class LoyaltyPlugin implements GatrionPlugin {
  @override
  String get name => 'loyalty';
  
  @override
  Future<void> initialize(GatrionSDK sdk) async {
    // Store reference to SDK for internal use
    _sdk = sdk;
    
    // Setup any required state
    _points = await _sdk.storage.secureRead('loyalty_points');
    if (_points == null) {
      _points = '0';
      await _sdk.storage.secureWrite('loyalty_points', '0');
    }
  }
  
  late GatrionSDK _sdk;
  String _points = '0';
  
  // Custom methods
  Future<int> getPoints() async {
    final response = await _sdk.client.get('/loyalty/points');
    _points = response['points'].toString();
    await _sdk.storage.secureWrite('loyalty_points', _points);
    return int.parse(_points);
  }
  
  Future<void> redeem(int points) async {
    await _sdk.client.post('/loyalty/redeem', body: {'points': points});
    await getPoints(); // refresh
  }
}`} lang="bash" title="loyalty_plugin.dart" />
</section>

<CodeBlock code={`// Register plugin during SDK initialization
await GatrionSDK.initialize(
  apiKey: 'YOUR_API_KEY',
  environment: Environment.sandbox,
  plugins: [
    LoyaltyPlugin(),
    AnalyticsPlugin(),
    ReferralPlugin(),
  ],
);

// Access plugin anywhere in the app
final loyalty = GatrionSDK.plugin<LoyaltyPlugin>();
final points = await loyalty.getPoints();
print('Loyalty points: $points');

await loyalty.redeem(100);
print('Redeemed! New balance: \${await loyalty.getPoints()}');`} lang="bash" title="register.dart" />
</section>

<CodeBlock code={`class MixpanelPlugin implements GatrionPlugin {
  @override
  String get name => 'mixpanel';
  
  late GatrionSDK _sdk;
  String? _distinctId;
  
  @override
  Future<void> initialize(GatrionSDK sdk) async {
    _sdk = sdk;
    // Initialize Mixpanel here
  }
  
  @override
  Future<void> onUserLogin(User user) async {
    _distinctId = user.id;
    // identify user in Mixpanel
  }
  
  @override
  Future<void> onUserLogout() async {
    _distinctId = null;
  }
  
  void track(String event, {Map<String, dynamic>? properties}) {
    final props = {
      ...?properties,
      'distinct_id': _distinctId,
      'app_version': _sdk.appVersion,
      'platform': Platform.operatingSystem,
    };
    // send to Mixpanel
    debugPrint('[Mixpanel] $event: $props');
  }
}

// Usage
final mixpanel = GatrionSDK.plugin<MixpanelPlugin>();
mixpanel.track('button_clicked', properties: {'button': 'add_to_cart'});`} lang="bash" title="mixpanel_plugin.dart" />

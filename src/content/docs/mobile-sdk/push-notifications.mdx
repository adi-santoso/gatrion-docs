final status = await GatrionSDK.notifications.requestPermission();

if (status == PermissionStatus.granted) {
  print('Notification permission granted');
  
  // Get FCM token
  final token = await GatrionSDK.notifications.getToken();
  print('FCM Token: $token');
  
  // Register token to backend
  await GatrionSDK.notifications.registerToken();
}`} lang="bash" title="permission.dart" />

await GatrionSDK.notifications.subscribeToTopic('promotions');

// Unsubscribe
await GatrionSDK.notifications.unsubscribeFromTopic('promotions');

// Subscribe based on user role
final user = await GatrionSDK.auth.getCurrentUser();
if (user != null) {
  await GatrionSDK.notifications.subscribeToTopic('user_\${user.id}');
  if (user.role == 'ADMIN') {
    await GatrionSDK.notifications.subscribeToTopic('admin_alerts');
  }
}`} lang="bash" title="topics.dart" />
</section>

<section class="animate-fade-up [animation-delay:.25s]">
<h2 id="deep-linking" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">

</h2>

<CodeBlock code={`// Notification payload with deep link
/*
  Server sends:
  {
    "notification": {
      "title": "Order Update",
      "body": "Your order is being shipped"
    },
    "data": {
      "type": "order_update",
      "orderId": "order_456",
      "deepLink": "gatrion://orders/order_456"
    }
  }
*/

// Handle deep link
GatrionSDK.notifications.onMessageOpened.listen((message) {
  final deepLink = message.data['deepLink'];
  if (deepLink != null) {
    GatrionSDK.deeplink.handle(deepLink);
  }
});

// Register deep link routes
GatrionSDK.deeplink.registerRoutes({
  'orders/:id': (params) => OrderDetailPage(orderId: params['id']!),
  'products/:id': (params) => ProductDetailPage(productId: params['id']!),
});`} lang="bash" title="deep-linking.dart" />
</section>

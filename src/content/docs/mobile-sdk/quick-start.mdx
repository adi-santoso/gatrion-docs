  <a href="#initialize" class="flex items-center gap-2">Initialize SDK <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<CodeBlock code={`import 'package:gatrion_sdk/gatrion_sdk.dart';
import 'package:flutter/material.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  await GatrionSDK.initialize(
    apiKey: 'YOUR_API_KEY',
    environment: Environment.sandbox, // or .production
    enableLogging: true,
  );

  runApp(MyApp());
}`} lang="bash" title="main.dart" />

<Callout type="warning">
  **Security:** Jangan hardcode API key. Gunakan <code class="rounded-md bg-amber-500/10 px-1.5 py-0.5 font-mono text-[12px] text-amber-600 dark:text-amber-300">--dart-define</code> atau <code class="rounded-md bg-amber-500/10 px-1.5 py-0.5 font-mono text-[12px] text-amber-600 dark:text-amber-300">flutter_dotenv</code>.
</Callout>
</section>

<CodeBlock code={`// Login
final result = await GatrionSDK.auth.login(
  email: 'user@example.com',
  password: 'password123',
);

if (result.success) {
  print('Welcome, \${result.user.name}!');
}

// Get products
final products = await GatrionSDK.products.list(
  page: 1,
  limit: 20,
);

for (final p in products.data) {
  print('\${p.name} - Rp \${p.price}');
}`} lang="bash" title="example.dart" />
</section>

<section class="animate-fade-up [animation-delay:.25s]">
<h2 id="platform-setup" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">

</h2>

<CodeBlock code={`<!-- Android: android/app/src/main/AndroidManifest.xml -->
<manifest>
  <uses-permission android:name="android.permission.INTERNET" />
  
  <application>
    <meta-data
      android:name="gatrion.api_key"
      android:value="YOUR_API_KEY" />
  </application>
</manifest>`} lang="bash" title="AndroidManifest.xml" />

<CodeBlock code={`// iOS: ios/Runner/Info.plist
<key>GATRION_API_KEY</key>
<string>YOUR_API_KEY</string>

// ios/Runner/AppDelegate.swift
import gatrion_sdk

func application(...) -> Bool {
  GatrionSDK.configure(apiKey: "YOUR_API_KEY")
  return true
}`} lang="bash" title="iOS Setup" />
</section>

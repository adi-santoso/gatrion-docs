import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';
import 'package:gatrion_sdk/gatrion_sdk.dart';

class MockAuth extends Mock implements GatrionAuth {}

void main() {
  late MockAuth mockAuth;

  setUp(() {
    mockAuth = MockAuth();
    registerFallbackValue(AuthResult.empty());
  });

  group('Auth', () {
    test('login returns success', () async {
      // arrange
      when(() => mockAuth.login(
        email: any(named: 'email'),
        password: any(named: 'password'),
      )).thenAnswer((_) async => AuthResult.success(
        user: User(id: '1', name: 'Test', email: 'test@test.com', role: 'USER'),
        token: 'mock-token',
      ));

      // act
      final result = await mockAuth.login(
        email: 'test@test.com',
        password: 'password',
      );

      // assert
      expect(result.success, true);
      expect(result.user.name, 'Test');
      verify(() => mockAuth.login(
        email: 'test@test.com',
        password: 'password',
      )).called(1);
    });

    test('login fails with invalid credentials', () async {
      when(() => mockAuth.login(
        email: any(named: 'email'),
        password: any(named: 'password'),
      )).thenThrow(GatrionException(
        type: GatrionErrorType.unauthorized,
        message: 'Invalid credentials',
      ));

      expect(
        () => mockAuth.login(email: 'wrong@test.com', password: 'wrong'),
        throwsA(isA<GatrionException>()),
      );
    });
  });
}`} lang="bash" title="auth_test.dart" />

import 'package:http/testing.dart';
import 'package:http/http.dart' as http;

void main() {
  setUpAll(() {
    // Mock HTTP client
    GatrionSDK.test.setMockClient(MockClient((request) async {
      if (request.url.path.contains('/users')) {
        return http.Response(
          jsonEncode({
            'statusCode': 200,
            'data': [
              {'id': '1', 'name': 'Test User', 'email': 'test@test.com'}
            ],
          }),
          200,
          headers: {'content-type': 'application/json'},
        );
      }
      
      if (request.url.path.contains('/auth/login')) {
        return http.Response(
          jsonEncode({
            'statusCode': 200,
            'data': {
              'accessToken': 'mock-token',
              'user': {'id': '1', 'name': 'Test', 'email': 'test@test.com'},
            },
          }),
          200,
        );
      }
      
      return http.Response('', 404);
    }));
  });

  test('getUsers returns list', () async {
    final users = await GatrionSDK.users.list();
    expect(users.data, hasLength(1));
    expect(users.data.first.name, 'Test User');
  });
}`} lang="bash" title="mock_http_test.dart" />

import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';

class MockAuth extends Mock implements GatrionAuth {}

void main() {
  testWidgets('Login form submits', (tester) async {
    final mockAuth = MockAuth();
    
    when(() => mockAuth.login(
      email: any(named: 'email'),
      password: any(named: 'password'),
    )).thenAnswer((_) async => AuthResult.success(
      user: User(id: '1', name: 'Test', email: 'test@test.com', role: 'USER'),
      token: 'token',
    ));

    await tester.pumpWidget(MaterialApp(
      home: LoginPage(auth: mockAuth),
    ));

    // Enter email
    await tester.enterText(
      find.byKey(const Key('email_field')),
      'test@test.com',
    );
    
    // Enter password
    await tester.enterText(
      find.byKey(const Key('password_field')),
      'password123',
    );

    // Tap login button
    await tester.tap(find.byKey(const Key('login_button')));
    await tester.pumpAndSettle();

    // Verify login was called
    verify(() => mockAuth.login(
      email: 'test@test.com',
      password: 'password123',
    )).called(1);
  });
}`} lang="bash" title="widget_test.dart" />

      await tester.pumpAndSettle();

      // Should see orders list

    });
  });

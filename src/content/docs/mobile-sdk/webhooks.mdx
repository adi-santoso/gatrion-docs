<CodeBlock code={`// Listen to specific event
GatrionSDK.webhooks.on('payment.settled', (data) {
  print('Payment settled: \${data['paymentId']}');
  refreshOrders();
  showSuccess('Payment successful!');
});

// Listen to multiple events
GatrionSDK.webhooks.on('payment.failed', (data) {
  showError('Payment failed for order \${data['orderId']}');
});

// Listen to all events
GatrionSDK.webhooks.onAll((event, data) {
  debugPrint('Event: $event, Data: $data');
  
  // Log to analytics
  GatrionSDK.analytics.track('webhook_received', properties: {
    'event': event,
    'payment_id': data['paymentId'],
  });
});`} lang="bash" title="listening.dart" />
</section>

<CodeBlock code={`// Handle webhook in background (via FCM background handler)
@pragma('vm:entry-point')
Future<void> backgroundHandler(RemoteMessage message) async {
  await GatrionSDK.initialize(
    apiKey: 'YOUR_API_KEY',
    environment: Environment.production,
  );

  final event = message.data['event'];
  final data = jsonDecode(message.data['data']);

  switch (event) {
    case 'payment.settled':
      // Update local database
      await GatrionSDK.offline.save('payments', data);
      // Show local notification
      await GatrionSDK.notifications.showLocal(
        title: 'Payment Successful',
        body: 'Your payment of Rp \${data['amount']} has been settled',
      );
      break;
    case 'order.status_changed':
      await GatrionSDK.offline.save('order_status', data);
      break;
  }
}

// Register in main()
void main() async {
  FirebaseMessaging.onBackgroundMessage(backgroundHandler);
  // ...
}`} lang="bash" title="background-handler.dart" />

<Callout type="warning">
  **Important:** Background handler harus top-level function dan ditandai dengan <code class="rounded-md bg-amber-500/10 px-1.5 py-0.5 font-mono text-[12px] text-amber-600 dark:text-amber-300">@pragma('vm:entry-point')</code>.
</Callout>

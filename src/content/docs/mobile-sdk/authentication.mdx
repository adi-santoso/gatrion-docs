<h2 id="biometric" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">

</h2>

<CodeBlock code={`// Check biometric availability
final isAvailable = await GatrionSDK.auth.isBiometricAvailable();
// returns: BiometricAvailability.available | unavailable | notEnrolled

if (isAvailable == BiometricAvailability.available) {
  // Enable biometric login
  await GatrionSDK.auth.enableBiometric();
  
  // Login with biometric
  final result = await GatrionSDK.auth.loginWithBiometric();
  if (result.success) {
    print('Biometric login successful!');
  }
}`} lang="bash" title="biometric.dart" />

<Callout type="info">
  **Platform Support:** Fingerprint (Android), Face ID (iOS), dan fingerprint (iOS). SDK menggunakan <code class="rounded-md bg-violet-500/10 px-1.5 py-0.5 font-mono text-[12px] text-violet-600 dark:text-violet-300">local_auth</code> package.
</Callout>

<h2 id="session" class="group scroll-mt-24 text-xl font-bold tracking-tight text-zinc-900 dark:text-white">
  <a href="#session" class="flex items-center gap-2">Session Management <span class="text-violet-400 opacity-0 transition group-hover:opacity-100">#</span></a>
</h2>

<CodeBlock code={`// Check if user is logged in
final isLoggedIn = await GatrionSDK.auth.isLoggedIn();

// Get current user
final user = await GatrionSDK.auth.getCurrentUser();
if (user != null) {
  print('Logged in as: \${user.name}');
}

// Listen to auth state changes
GatrionSDK.auth.authStateChanges.listen((state) {
  if (state == AuthState.loggedOut) {
    // redirect to login
    Navigator.pushReplacementNamed(context, '/login');
  }
});

// Logout
await GatrionSDK.auth.logout();
// clears token, navigate to login screen`} lang="bash" title="session.dart" />
</section>

});

final newToken = await GatrionSDK.auth.refreshToken();`} lang="bash" title="token-refresh.dart" />
